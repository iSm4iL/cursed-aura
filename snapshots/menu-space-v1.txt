СНИМОК КОСМИЧЕСКОГО ГЛАВНОГО МЕНЮ — версия 1.0.19, тег menu-space-v1
===================================================================
Это ЗАМОРОЖЕННАЯ КОПИЯ для справки, а не рабочий код. Игра её не читает.

Полный откат к этому состоянию:
    git checkout menu-space-v1 -- src/game.html && npm run build

Посмотреть, не откатывая:
    git show menu-space-v1:src/game.html

Что тут лежит: модуль сцены (генерация мира, попиксельная планета, камера,
метки, глаза, поверхность), разметка меню и CSS-шкура интерфейса.
Переиспользовать при переходе на тронный зал стоит: машинку камеры
(ступени + smoothstep), пиксельный буфер с nearest-neighbour, DOM-панели,
слой мета-прогрессии и саму шкуру интерфейса.

/* ===== JS: модуль сцены (sky*) ===== */
  // ====== Космическая сцена главного меню ====================================
  // Canvas под DOM-панелями меню со своим rAF-циклом. Цикл крутится ТОЛЬКО пока
  // меню открыто — во время забега остановлен и не отъедает ни кадра у Phaser.
  // Камера — одно число z: 0 — вид из космоса, 1 — орбита (на глобусе горят метки
  // локаций), 2 — над самой поверхностью (горизонт и силуэты руин). Композиция
  // каждой ступени задана руками (SKY_STAGES), между ними smoothstep — за счёт
  // этого наезд читается как полёт к планете, а не как смена картинки.
  const SKY_REDUCED = matchMedia('(prefers-reduced-motion: reduce)').matches;
  const SKY_TILT = -0.24;                       // наклон оси планеты, рад
  const SKY_STAGES = [
    // r/cx/cy — доли от min(w,h) и от размеров экрана. Колонки W/N — широкий экран
    // и узкий: на широком планета уезжает влево под панель справа, на узком уходит
    // вверх, а панель садится вниз (см. парный media-query в CSS).
    { rW: 0.38, rN: 0.38, cxW: 0.50, cxN: 0.50, cyW: 1.13, cyN: 1.15, star: 1.00, pin: 0.00, surf: 0 },
    { rW: 0.46, rN: 0.46, cxW: 0.33, cxN: 0.50, cyW: 0.52, cyN: 0.30, star: 1.30, pin: 1.00, surf: 0 },
    // Последняя ступень считается от ВЫСОТЫ экрана, а не от min(w,h): на вытянутом
    // телефоне радиус от ширины оставлял горизонт посреди экрана вместо земли под ногами.
    { rW: 4.00, rN: 8.60, cxW: 0.40, cxN: 0.50, cyW: 4.30, cyN: 4.30, star: 1.85, pin: 0.00, surf: 1 }
  ];
  const sky = {
    // w/h — размер буфера (в «крупных пикселях»), cssW/cssH — размер на экране.
    canvas: null, ctx: null, out: null, buf: null, bctx: null,
    w: 0, h: 0, cssW: 0, cssH: 0, pix: 3, raf: 0, last: 0, t: 0,
    z: 0, zTarget: 0, spin: 0, spinTarget: null,
    stars: [], land: [], clouds: [], skyline: [], ready: false
  };

  function hexToRgba(hex, a){
    const n = parseInt(hex.slice(1), 16);
    return 'rgba(' + ((n >> 16) & 255) + ',' + ((n >> 8) & 255) + ',' + (n & 255) + ',' + a + ')';
  }

  // Сцена рисуется в МАЛЕНЬКИЙ буфер и растягивается без сглаживания — отсюда
  // пиксельная подача из референсов (refs/ref_space.jpg, ref_planet.jpg). Вся
  // отрисовка ниже работает в координатах буфера, поэтому менять её не пришлось:
  // sky.w/h — это размер буфера, а sky.cssW/cssH — реальный размер на экране.
  function skyResize(){
    if (!sky.canvas) return;
    sky.cssW = sky.canvas.clientWidth || window.innerWidth;
    sky.cssH = sky.canvas.clientHeight || window.innerHeight;
    // Блок в 3px давал слишком грубую сетку («вижу сетку») — оставляем 2,
    // на узком экране 1.5 нельзя, поэтому там честная единица.
    sky.pix = sky.cssW < 520 ? 1 : 2;
    sky.w = Math.ceil(sky.cssW / sky.pix);
    sky.h = Math.ceil(sky.cssH / sky.pix);
    if (!sky.buf){
      sky.buf = document.createElement('canvas');
      sky.bctx = sky.buf.getContext('2d');
    }
    sky.buf.width = sky.w;
    sky.buf.height = sky.h;
    sky.ctx = sky.bctx;                             // весь рисунок идёт в буфер
    sky.canvas.width = sky.cssW;
    sky.canvas.height = sky.cssH;
    sky.out = sky.canvas.getContext('2d');
    sky.out.imageSmoothingEnabled = false;
  }

  function skyInit(){
    const c = document.getElementById('menuSky');
    if (!c) return;
    sky.canvas = c;
    skyResize();
    const rnd = (a, b) => a + Math.random() * (b - a);
    // Звёзды хранятся в полярных координатах от центра экрана — тогда «разлёт»
    // при наезде это просто домножение радиуса, без пересчёта всего поля.
    sky.stars = [];
    const starCount = Math.max(90, Math.min(320, Math.round(sky.w * sky.h / 320)));
    for (let i = 0; i < starCount; i++){
      sky.stars.push({
        a: rnd(0, Math.PI * 2), d: Math.pow(Math.random(), 0.62),
        depth: rnd(0.3, 1), size: rnd(0.5, 1.9),
        warm: Math.random() < 0.22, tw: rnd(0, Math.PI * 2), tws: rnd(0.5, 2.2)
      });
    }
    skyBuildWorld();
    // Силуэт руин на горизонте — проявляется на последней ступени наезда.
    sky.skyline = [];
    for (let x = -0.05; x < 1.05; ){
      const bw = rnd(0.03, 0.09);
      sky.skyline.push({ x, w: bw, h: rnd(0.05, 0.2), broken: Math.random() < 0.6 });
      x += bw + rnd(0.005, 0.05);
    }
    sky.ready = true;
  }

  // ---- Мир планеты -----------------------------------------------------------
  // Карта строится ОДИН РАЗ при запуске: трёхмерный шум прямо на сфере (а не на
  // прямоугольнике), поэтому нет шва по нулевому меридиану и нет каши на полюсах.
  // Хранится равнопромежуточной картой WORLD_W x WORLD_H: цвет суши/океана, маска
  // облаков и маска проклятых зон. Дальше она попиксельно натягивается на диск.
  const WORLD_W = 320, WORLD_H = 160;
  const SEA = 0.505;                                   // порог «суша/вода»
  function skyHash3(x, y, z){
    let n = Math.sin(x * 127.1 + y * 311.7 + z * 74.7) * 43758.5453;
    return n - Math.floor(n);
  }
  function skyNoise3(x, y, z){
    const xi = Math.floor(x), yi = Math.floor(y), zi = Math.floor(z);
    const xf = x - xi, yf = y - yi, zf = z - zi;
    const u = xf * xf * (3 - 2 * xf), v = yf * yf * (3 - 2 * yf), w = zf * zf * (3 - 2 * zf);
    const c = (dx, dy, dz) => skyHash3(xi + dx, yi + dy, zi + dz);
    const x00 = c(0,0,0) + (c(1,0,0) - c(0,0,0)) * u;
    const x10 = c(0,1,0) + (c(1,1,0) - c(0,1,0)) * u;
    const x01 = c(0,0,1) + (c(1,0,1) - c(0,0,1)) * u;
    const x11 = c(0,1,1) + (c(1,1,1) - c(0,1,1)) * u;
    return (x00 + (x10 - x00) * v) + ((x01 + (x11 - x01) * v) - (x00 + (x10 - x00) * v)) * w;
  }
  function skyFbm(x, y, z, oct, lac, gain){
    let a = 0.5, f = 1, sum = 0, norm = 0;
    for (let i = 0; i < oct; i++){
      sum += a * skyNoise3(x * f, y * f, z * f);
      norm += a; a *= (gain || 0.5); f *= (lac || 2.07);
    }
    return sum / norm;
  }
  function skyBuildWorld(){
    const W = WORLD_W, H = WORLD_H;
    sky.world = { w: W, h: H, rgb: new Uint8Array(W * H * 3), cloud: new Uint8Array(W * H),
                  curse: new Uint8Array(W * H), land: new Uint8Array(W * H) };
    const wd = sky.world;
    // Палитра нарочно приглушённая: по арт-библии мир тёмный, цвет отдан магии.
    const OCEAN_DEEP = [10, 24, 42], OCEAN = [17, 44, 68], SHALLOW = [28, 66, 88];
    const SAND = [58, 58, 44], GRASS = [32, 56, 38], FOREST = [22, 42, 28];
    const ROCK = [56, 54, 48], ICE = [150, 162, 170];
    const seedA = Math.random() * 100, seedB = Math.random() * 100;
    for (let v = 0; v < H; v++){
      const lat = (90 - (v + 0.5) * 180 / H) * Math.PI / 180;
      const cl = Math.cos(lat), sl = Math.sin(lat);
      for (let u = 0; u < W; u++){
        const lon = ((u + 0.5) * 360 / W) * Math.PI / 180;
        const px = cl * Math.sin(lon), py = sl, pz = cl * Math.cos(lon);
        // Континенты: низкая частота задаёт массивы, высокая рвёт береговую линию.
        let hgt = skyFbm(px * 1.9 + seedA, py * 1.9, pz * 1.9 + seedA, 5);
        hgt = hgt * 0.78 + skyFbm(px * 5.5, py * 5.5, pz * 5.5, 3) * 0.22;
        hgt -= Math.pow(Math.abs(sl), 3) * 0.10;        // полюса чуть ниже: меньше суши
        const i = v * W + u, o = i * 3;
        let col;
        if (hgt > SEA){
          const t = (hgt - SEA) / (1 - SEA);            // 0 у берега, 1 в горах
          if (t < 0.06) col = SAND;
          else if (t < 0.30) col = GRASS;
          else if (t < 0.62) col = FOREST;
          else col = ROCK;
          wd.land[i] = 1;
        } else {
          const t = (SEA - hgt) / SEA;
          col = t < 0.10 ? SHALLOW : (t < 0.35 ? OCEAN : OCEAN_DEEP);
        }
        // Ледяные шапки поверх всего у полюсов, с рваной границей.
        const ice = Math.abs(sl) - 0.80 + skyFbm(px * 7 + 40, py * 7, pz * 7, 2) * 0.16;
        if (ice > 0) col = ICE;
        wd.rgb[o] = col[0]; wd.rgb[o+1] = col[1]; wd.rgb[o+2] = col[2];
        // Облака — своя сфера шума, крутится отдельно от суши.
        const cd = skyFbm(px * 2.6 + seedB, py * 2.6, pz * 2.6 + seedB, 4);
        wd.cloud[i] = Math.max(0, Math.min(255, Math.round((cd - 0.50) * 900)));
        // Проклятые зоны: редкие очаги строго на суше — оттуда и горит.
        const cu = skyFbm(px * 1.5 - 17, py * 1.5 + 9, pz * 1.5 - 17, 3);
        wd.curse[i] = wd.land[i] && cu > 0.60 ? Math.min(255, Math.round((cu - 0.60) * 1100)) : 0;
      }
    }
    // Локации обязаны стоять на суше — раньше метки оказывались посреди океана.
    skyPlaceLocations();
    // Глаза открываются над самыми сильными очагами скверны.
    sky.eyes = skyPickEyes(3);
  }
  // Ищет для каждой локации ближайшую к её «желаемой» точке клетку суши.
  function skyPlaceLocations(){
    const wd = sky.world, W = wd.w, H = wd.h;
    LOCATIONS.forEach((loc) => {
      let best = null, bestD = 1e9;
      for (let v = 4; v < H - 4; v++){
        const lat = 90 - (v + 0.5) * 180 / H;
        if (Math.abs(lat) > 66) continue;              // не селим во льдах
        for (let u = 0; u < W; u++){
          const i = v * W + u;
          if (!wd.land[i] || wd.curse[i] > 40) continue;
          const lon = (u + 0.5) * 360 / W - 180;
          const dLon = ((lon - loc.lon + 540) % 360) - 180;
          const d = dLon * dLon + (lat - loc.lat) * (lat - loc.lat) * 2.2;
          if (d < bestD){ bestD = d; best = { lat, lon }; }
        }
      }
      if (best){ loc.lat = best.lat; loc.lon = best.lon; }
    });
  }
  function skyPickEyes(n){
    const wd = sky.world, W = wd.w, H = wd.h, out = [];
    const cand = [];
    for (let v = 10; v < H - 10; v += 2){
      for (let u = 0; u < W; u += 2){
        const i = v * W + u;
        if (wd.curse[i] > 150) cand.push({ lat: 90 - v * 180 / H, lon: u * 360 / W - 180, s: wd.curse[i] });
      }
    }
    cand.sort((a, b) => b.s - a.s);
    cand.forEach((c) => {
      if (out.length >= n) return;
      // Не лепим глаза друг на друга.
      if (out.some((o) => Math.abs(o.lat - c.lat) < 28 && Math.abs(((o.lon - c.lon + 540) % 360) - 180) < 40)) return;
      out.push({ lat: c.lat, lon: c.lon, phase: Math.random() * Math.PI * 2, open: 0 });
    });
    return out;
  }

  // lat/lon на единичной сфере с наклоном оси. z > 0 — точка на видимой стороне.
  function skyProject(lat, lon){
    const la = lat * Math.PI / 180, lo = (lon + sky.spin) * Math.PI / 180;
    const x = Math.cos(la) * Math.sin(lo);
    const y0 = Math.sin(la), z0 = Math.cos(la) * Math.cos(lo);
    const c = Math.cos(SKY_TILT), s = Math.sin(SKY_TILT);
    return { x, y: y0 * c - z0 * s, z: y0 * s + z0 * c };
  }

  function skyCam(){
    // Считаем по ЭКРАНУ, а не по буферу: буфер вдвое меньше, и десктоп 1280px
    // проваливался в узкую раскладку, пока CSS-медиазапрос считал его широким —
    // отсюда и разъехавшаяся композиция.
    const wide = sky.cssW > 760;
    const S = Math.min(sky.w, sky.h);
    const i = Math.max(0, Math.min(SKY_STAGES.length - 1, Math.floor(sky.z)));
    const j = Math.min(SKY_STAGES.length - 1, i + 1);
    const f0 = Math.max(0, Math.min(1, sky.z - i));
    const f = f0 * f0 * (3 - 2 * f0);                       // smoothstep
    const A = SKY_STAGES[i], B = SKY_STAGES[j];
    const mix = (a, b) => a + (b - a) * f;
    return {
      R: mix(wide ? A.rW : A.rN, wide ? B.rW : B.rN) * S,
      cx: mix(wide ? A.cxW : A.cxN, wide ? B.cxW : B.cxN) * sky.w,
      cy: mix(wide ? A.cyW : A.cyN, wide ? B.cyW : B.cyN) * sky.h,
      star: mix(A.star, B.star), pin: mix(A.pin, B.pin), surf: mix(A.surf, B.surf)
    };
  }

  function skyDrawStars(ctx, cam){
    const w = sky.w, h = sky.h;
    const maxR = Math.hypot(w, h) * 0.62;
    // У поверхности звёзды не гасим полностью — над руинами всё ещё ночное небо.
    const fade = Math.max(0.4, 1 - cam.surf * 0.6);
    sky.stars.forEach((s) => {
      const push = 1 + (cam.star - 1) * s.depth * 1.7;      // ближние разлетаются сильнее
      const d = s.d * maxR * push;
      const x = w * 0.5 + Math.cos(s.a) * d;
      const y = h * 0.5 + Math.sin(s.a) * d * 0.84;
      if (x < -4 || x > w + 4 || y < -4 || y > h + 4) return;
      const tw = SKY_REDUCED ? 0.85 : 0.55 + 0.45 * Math.sin(sky.t * s.tws + s.tw);
      ctx.globalAlpha = tw * fade * (0.3 + s.depth * 0.7);
      ctx.fillStyle = s.warm ? '#ffd9b0' : '#dfe9ff';
      const sz = s.size * (0.8 + s.depth * 0.6);
      ctx.fillRect(x, y, sz, sz);
    });
    ctx.globalAlpha = 1;
  }

  // Чёрная дыра: горизонт событий + аккреционный диск, ближняя кромка которого
  // проходит ПЕРЕД диском — тот самый узнаваемый силуэт с линзированием.
  function skyDrawBlackHole(ctx, x, y, r, fade){
    if (fade <= 0.01) return;
    ctx.save();
    ctx.globalAlpha = fade;
    ctx.translate(x, y);
    ctx.rotate(-0.32);
    const glow = ctx.createRadialGradient(0, 0, r * 0.9, 0, 0, r * 4.4);
    glow.addColorStop(0, 'rgba(255,196,120,0.26)');
    glow.addColorStop(0.45, 'rgba(200,90,60,0.09)');
    glow.addColorStop(1, 'rgba(0,0,0,0)');
    ctx.fillStyle = glow;
    ctx.fillRect(-r * 4.4, -r * 4.4, r * 8.8, r * 8.8);
    const ring = (sy, lw, col) => {
      ctx.save();
      ctx.scale(1, sy);
      ctx.lineWidth = lw; ctx.strokeStyle = col;
      ctx.beginPath(); ctx.arc(0, 0, r * 1.95, 0, Math.PI * 2); ctx.stroke();
      ctx.restore();
    };
    ring(1, r * 0.26, 'rgba(255,150,70,0.45)');             // гало над горизонтом
    ring(0.2, r * 0.42, 'rgba(255,205,140,0.85)');          // диск почти с ребра
    ctx.fillStyle = '#000';
    ctx.beginPath(); ctx.arc(0, 0, r, 0, Math.PI * 2); ctx.fill();
    ctx.save();
    ctx.beginPath(); ctx.rect(-r * 3, 0, r * 6, r * 3); ctx.clip();
    ring(0.2, r * 0.42, 'rgba(255,222,168,0.95)');
    ctx.restore();
    ctx.restore();
  }

  function skyDrawSun(ctx, x, y, r, fade){
    if (fade <= 0.01) return;
    // Дыхание короны + лёгкая пульсация самого диска: звёзды мерцают, а солнце
    // стояло мёртвым пятном — на это была прямая жалоба.
    const puls = SKY_REDUCED ? 1 : 1 + Math.sin(sky.t * 1.7) * 0.10 + Math.sin(sky.t * 4.3) * 0.04;
    r *= puls;
    const g = ctx.createRadialGradient(x, y, 0, x, y, r * 9);
    g.addColorStop(0, 'rgba(255,246,220,' + (0.95 * fade) + ')');
    g.addColorStop(0.07, 'rgba(255,214,140,' + (0.5 * fade) + ')');
    g.addColorStop(0.3, 'rgba(255,150,80,' + (0.12 * fade) + ')');
    g.addColorStop(1, 'rgba(255,120,60,0)');
    ctx.fillStyle = g;
    ctx.fillRect(x - r * 9, y - r * 9, r * 18, r * 18);
    ctx.globalAlpha = fade;
    ctx.fillStyle = '#fff6e0';
    ctx.beginPath(); ctx.arc(x, y, r, 0, Math.PI * 2); ctx.fill();
    ctx.globalAlpha = 1;
  }

  // Общий приём для шара: рисуем плоско, потом кладём сверху ОДИН радиальный
  // градиент от точки освещения — он же даёт терминатор и тёмный лимб.
  function skyShade(ctx, cx, cy, R, lit, strength){
    const g = ctx.createRadialGradient(cx + lit.x * R * 0.6, cy + lit.y * R * 0.6, R * 0.05, cx, cy, R * 1.15);
    g.addColorStop(0, 'rgba(255,240,215,' + (0.22 * strength) + ')');
    g.addColorStop(0.35, 'rgba(255,220,190,' + (0.05 * strength) + ')');
    g.addColorStop(0.62, 'rgba(2,4,10,' + (0.45 * strength) + ')');
    g.addColorStop(1, 'rgba(0,1,5,' + (0.93 * strength) + ')');
    ctx.fillStyle = g;
    ctx.fillRect(cx - R, cy - R, R * 2, R * 2);
  }

  // Планета натягивается на диск ПОПИКСЕЛЬНО: для каждого пикселя восстанавливаем
  // точку на сфере, снимаем наклон оси и вращение, берём цвет из карты мира. Так
  // получаются настоящие материки с береговой линией, а не пятна, и вращение
  // выглядит как вращение шара, а не как ползающие кляксы.
  function skyDrawPlanet(ctx, cx, cy, R, lit, cam){
    const atmT = Math.min(R * 0.32, Math.min(sky.w, sky.h) * 0.16);
    const atmA = 1 - cam.surf * 0.75;
    const atm = ctx.createRadialGradient(cx, cy, Math.max(0, R - atmT * 0.2), cx, cy, R + atmT);
    atm.addColorStop(0, 'rgba(90,150,190,0)');
    atm.addColorStop(0.22, 'rgba(90,160,200,' + (0.26 * atmA) + ')');
    atm.addColorStop(1, 'rgba(60,110,170,0)');
    ctx.fillStyle = atm;
    ctx.beginPath(); ctx.arc(cx, cy, R + atmT, 0, Math.PI * 2); ctx.fill();

    const wd = sky.world;
    // У самой поверхности карта растянулась бы на несколько текселей во весь экран —
    // там рисуем землю, а не глобус (см. skyDrawGround).
    const mapAlpha = 1 - Math.min(1, Math.max(0, (cam.surf - 0.35) / 0.45));
    if (wd && mapAlpha > 0.01){
      const x0 = Math.max(0, Math.floor(cx - R)), x1 = Math.min(sky.w, Math.ceil(cx + R));
      const y0 = Math.max(0, Math.floor(cy - R)), y1 = Math.min(sky.h, Math.ceil(cy + R));
      if (x1 > x0 && y1 > y0){
        const img = ctx.getImageData(x0, y0, x1 - x0, y1 - y0);
        const px = img.data;
        const W = wd.w, H = wd.h;
        const ct = Math.cos(SKY_TILT), st = Math.sin(SKY_TILT);
        const spinU = ((-sky.spin / 360) % 1 + 1) % 1;
        const cloudU = ((-(sky.spin * 1.35 + sky.t * 4) / 360) % 1 + 1) % 1;
        // Свет в экранных координатах: z>0 — на зрителя.
        const lz = 0.42;
        const ll = Math.hypot(lit.x, lit.y, lz);
        const lx = lit.x / ll, ly = lit.y / ll, lzz = lz / ll;
        const flick = 0.72 + 0.28 * Math.sin(sky.t * 6.1);
        const invR = 1 / R;
        // Вблизи диск занимает весь буфер, и честный проход по каждому пикселю
        // стоил бы десятки миллисекунд на телефоне. Там карта и так растянута,
        // поэтому считаем через пиксель и заливаем блоками — на глаз не отличить.
        const step = R > Math.min(sky.w, sky.h) * 0.9 ? 2 : 1;
        const rowW = x1 - x0;
        for (let y = y0; y < y1; y += step){
          const ny = (y - cy) * invR;
          const rowBase = (y - y0) * rowW * 4;
          for (let x = x0; x < x1; x += step){
            const nx = (x - cx) * invR;
            const r2 = nx * nx + ny * ny;
            if (r2 > 1) continue;
            const nz = Math.sqrt(1 - r2);
            // Снимаем наклон оси: обратный поворот вокруг X.
            const wy0 = -ny, wz0 = nz;
            const wy = wy0 * ct + wz0 * st;
            const wz = -wy0 * st + wz0 * ct;
            const lat = Math.asin(Math.max(-1, Math.min(1, wy)));
            const lon = Math.atan2(nx, wz);
            let u = lon * 0.15915494 + 0.5 + spinU;      // 1/(2pi)
            u -= Math.floor(u);
            const v = 0.5 - lat * 0.31830989;            // 1/pi
            const mu = (u * W) | 0;
            const mv = Math.min(H - 1, Math.max(0, (v * H) | 0));
            const mi = mv * W + mu;
            const mo = mi * 3;
            let cr = wd.rgb[mo], cg = wd.rgb[mo + 1], cb = wd.rgb[mo + 2];
            // Облака — своя развёртка, поэтому ползут относительно суши.
            let cu = u + cloudU; cu -= Math.floor(cu);
            const ci = mv * W + ((cu * W) | 0);
            const cl = wd.cloud[ci] / 255;
            if (cl > 0){
              const k = cl * 0.85;
              cr += (205 - cr) * k; cg += (216 - cg) * k; cb += (228 - cb) * k;
            }
            // Освещение: одна честная диффузная составляющая + мягкий терминатор.
            let d = nx * lx + ny * ly + nz * lzz;
            d = d < 0 ? 0 : d;
            const shade = 0.10 + 0.95 * Math.pow(d, 0.85);
            cr *= shade; cg *= shade; cb *= shade;
            // Скверна горит и на ночной стороне — очаги видно даже в тени.
            const cv = wd.curse[mi];
            if (cv > 0){
              const k = (cv / 255) * flick;
              cr += (255 - cr) * k * 0.85;
              cg += (110 - cg) * k * 0.75;
              cb += (40 - cb) * k * 0.7;
            }
            for (let sy = 0; sy < step; sy++){
              if (y + sy >= y1) break;
              for (let sx = 0; sx < step; sx++){
                if (x + sx >= x1) break;
                const o = rowBase + sy * rowW * 4 + (x - x0 + sx) * 4;
                if (mapAlpha >= 0.999){
                  px[o] = cr; px[o+1] = cg; px[o+2] = cb; px[o+3] = 255;
                } else {
                  px[o]   += (cr - px[o]) * mapAlpha;
                  px[o+1] += (cg - px[o+1]) * mapAlpha;
                  px[o+2] += (cb - px[o+2]) * mapAlpha;
                  px[o+3] = 255;
                }
              }
            }
          }
        }
        ctx.putImageData(img, x0, y0);
      }
    }

    // Земля вблизи + свечение горизонта в цвете локации.
    if (cam.surf > 0.01) skyDrawGround(ctx, cx, cy, R, cam.surf);

    // Световая кромка на освещённой стороне — последний штрих объёма.
    const la = Math.atan2(lit.y, lit.x);
    const rim = ctx.createLinearGradient(cx + Math.cos(la) * R, cy + Math.sin(la) * R,
                                         cx - Math.cos(la) * R, cy - Math.sin(la) * R);
    rim.addColorStop(0, 'rgba(200,235,255,0.8)');
    rim.addColorStop(0.5, 'rgba(120,180,220,0.10)');
    rim.addColorStop(1, 'rgba(120,180,220,0)');
    ctx.strokeStyle = rim;
    ctx.lineWidth = Math.max(1, Math.min(R * 0.012, Math.min(sky.w, sky.h) * 0.010));
    ctx.globalAlpha = Math.max(0, 1 - cam.surf * 0.8);
    ctx.beginPath(); ctx.arc(cx, cy, R * 0.995, 0, Math.PI * 2); ctx.stroke();
    ctx.globalAlpha = 1;
  }

  // Земля на последней ступени: блочная текстура в цвете локации вместо пустой
  // заливки — раньше на этом месте был ровный серый экран.
  function skyDrawGround(ctx, cx, cy, R, k){
    const loc = LOCATIONS[curLocation] || LOCATIONS[0];
    const hy = cy - R;
    ctx.save();
    ctx.globalAlpha = Math.min(1, k);
    ctx.beginPath(); ctx.arc(cx, cy, R, 0, Math.PI * 2); ctx.clip();
    ctx.fillStyle = '#0c0f12';
    ctx.fillRect(cx - R, cy - R, R * 2, R * 2);
    const gnd = ctx.createLinearGradient(0, hy, 0, hy + sky.h);
    gnd.addColorStop(0, hexToRgba(loc.tint, 0.40));
    gnd.addColorStop(0.45, hexToRgba(loc.tint, 0.10));
    gnd.addColorStop(1, 'rgba(8,10,12,0)');
    ctx.fillStyle = gnd;
    ctx.fillRect(cx - R, cy - R, R * 2, R * 2);
    // Блочные пятна: даже слабая текстура снимает ощущение «пустой заливки».
    const blk = Math.max(2, Math.round(Math.min(sky.w, sky.h) / 90));
    const cols = Math.ceil(sky.w / blk), rows = Math.ceil((sky.h - hy) / blk);
    for (let i = 0; i < cols * rows * 0.22; i++){
      const bx = Math.floor(Math.random() * cols) * blk;
      const by = hy + Math.floor(Math.random() * rows) * blk;
      if (by < hy) continue;
      const depth = Math.min(1, Math.max(0, (by - hy) / Math.max(1, sky.h - hy)));
      ctx.fillStyle = Math.random() < 0.5
        ? 'rgba(0,0,0,' + (0.10 + depth * 0.18) + ')'
        : hexToRgba(loc.tint, 0.05 + (1 - depth) * 0.06);
      ctx.fillRect(bx, by, blk, blk);
    }
    ctx.restore();
  }

  // Проклятый глаз над очагом скверны: открывается, смотрит и закрывается.
  // Рисуется поверх диска, а не в попиксельном проходе — так им можно управлять
  // как отдельным существом, а не как текстурой.
  function skyDrawEyes(ctx, cx, cy, R, fade){
    if (!sky.eyes || fade <= 0.01) return;
    sky.eyes.forEach((e) => {
      const p = skyProject(e.lat, e.lon);
      if (p.z <= 0.12) return;
      // Цикл: долго закрыт, недолго открыт.
      const cyc = (Math.sin(sky.t * 0.42 + e.phase) + 1) / 2;
      const open = Math.max(0, (cyc - 0.55) / 0.45);
      if (open <= 0.02) return;
      const sx = cx + p.x * R, sy = cy - p.y * R;
      const s = R * 0.17 * (0.45 + p.z * 0.55);
      const a = fade * open * Math.min(1, p.z * 1.6);
      ctx.save();
      ctx.globalAlpha = a;
      ctx.translate(sx, sy);
      ctx.rotate(Math.atan2(sy - cy, sx - cx));
      ctx.scale(Math.max(0.18, p.z), 1);                 // ракурс на сфере
      const gl = ctx.createRadialGradient(0, 0, 0, 0, 0, s * 2.6);
      gl.addColorStop(0, 'rgba(255,140,60,0.55)');
      gl.addColorStop(1, 'rgba(255,80,20,0)');
      ctx.fillStyle = gl;
      ctx.fillRect(-s * 2.6, -s * 2.6, s * 5.2, s * 5.2);
      // Веко-миндалина: два дуговых сегмента, высота зависит от открытия.
      const hgt = s * 0.62 * open;
      ctx.beginPath();
      ctx.moveTo(-s, 0);
      ctx.quadraticCurveTo(0, -hgt * 2, s, 0);
      ctx.quadraticCurveTo(0, hgt * 2, -s, 0);
      ctx.closePath();
      ctx.fillStyle = '#120704';
      ctx.fill();
      ctx.strokeStyle = 'rgba(255,150,70,0.8)';
      ctx.lineWidth = Math.max(1, s * 0.09);
      ctx.stroke();
      // Зрачок смотрит чуть в сторону — глаз «живой».
      const look = Math.sin(sky.t * 0.7 + e.phase) * s * 0.22;
      ctx.beginPath();
      ctx.ellipse(look, 0, s * 0.20, hgt * 0.85, 0, 0, Math.PI * 2);
      ctx.fillStyle = '#ffb23c';
      ctx.fill();
      ctx.beginPath();
      ctx.ellipse(look, 0, s * 0.08, hgt * 0.7, 0, 0, Math.PI * 2);
      ctx.fillStyle = '#1a0602';
      ctx.fill();
      ctx.restore();
    });
  }

  function skyDrawMoon(ctx, cx, cy, R, lit, fade){
    if (fade <= 0.01) return;
    ctx.save();
    ctx.globalAlpha = fade;
    const g = ctx.createLinearGradient(cx - R, cy - R, cx + R, cy + R);
    g.addColorStop(0, '#9aa0ad');
    g.addColorStop(1, '#4e5460');
    ctx.beginPath(); ctx.arc(cx, cy, R, 0, Math.PI * 2);
    ctx.save(); ctx.clip();
    ctx.fillStyle = g;
    ctx.fillRect(cx - R, cy - R, R * 2, R * 2);
    ctx.fillStyle = 'rgba(0,0,0,0.18)';                     // пара кратеров
    ctx.beginPath(); ctx.arc(cx - R * 0.3, cy - R * 0.2, R * 0.26, 0, Math.PI * 2); ctx.fill();
    ctx.beginPath(); ctx.arc(cx + R * 0.35, cy + R * 0.3, R * 0.18, 0, Math.PI * 2); ctx.fill();
    skyShade(ctx, cx, cy, R, lit, 1);
    ctx.restore();
    ctx.restore();
  }

  // Метки локаций на глобусе. Ромб — тот же мотив, что у свотча класса в панели:
  // одна форма на весь экран выбора, а не два разных значка про одно и то же.
  function skyDrawPins(ctx, cx, cy, R, alpha){
    if (alpha <= 0.01) return;
    LOCATIONS.forEach((loc, i) => {
      const p = skyProject(loc.lat, loc.lon);
      if (p.z <= 0.05) return;
      const sel = i === curLocation;
      const col = loc.locked ? '#78809c' : loc.tint;
      const a = alpha * (0.25 + p.z * 0.75) * (sel ? 1 : 0.7);
      const s = Math.max(4, R * 0.03 * (0.6 + p.z * 0.4) * (sel ? 1.35 : 1));
      const sx = cx + p.x * R, sy = cy - p.y * R;
      ctx.save();
      ctx.globalAlpha = a;
      if (!loc.locked){                                      // луч вверх у доступной
        const beam = ctx.createLinearGradient(sx, sy, sx, sy - R * 0.45);
        beam.addColorStop(0, hexToRgba(col, 0.75));
        beam.addColorStop(1, hexToRgba(col, 0));
        ctx.fillStyle = beam;
        ctx.fillRect(sx - s * 0.3, sy - R * 0.45, s * 0.6, R * 0.45);
      }
      if (sel){
        const pulse = 1 + (SKY_REDUCED ? 0 : Math.sin(sky.t * 3) * 0.22);
        ctx.globalAlpha = a * 0.55;
        ctx.strokeStyle = col;
        ctx.lineWidth = Math.max(1, s * 0.16);
        ctx.beginPath(); ctx.arc(sx, sy, s * 2.1 * pulse, 0, Math.PI * 2); ctx.stroke();
        ctx.globalAlpha = a;
      }
      ctx.translate(sx, sy);
      ctx.rotate(Math.PI / 4);
      ctx.shadowColor = col;
      ctx.shadowBlur = s * 2.4;
      ctx.fillStyle = col;
      ctx.fillRect(-s / 2, -s / 2, s, s);
      ctx.restore();
    });
  }

  // Последняя ступень: свечение над горизонтом и обломанный силуэт руин, стоящий
  // НА линии горизонта (hy = верх сферы), а не у нижней кромки экрана — у нижней
  // его закрывала панель выбора класса, да и стоять ему физически негде.
  function skyDrawSurface(ctx, k, hy){
    if (k <= 0.01) return;
    const w = sky.w, h = sky.h;
    const loc = LOCATIONS[curLocation] || LOCATIONS[0];
    ctx.save();
    ctx.globalAlpha = k;
    const glow = ctx.createLinearGradient(0, hy - h * 0.3, 0, hy);
    glow.addColorStop(0, 'rgba(0,0,0,0)');
    glow.addColorStop(1, hexToRgba(loc.tint, 0.22));
    ctx.fillStyle = glow;
    ctx.fillRect(0, hy - h * 0.3, w, h * 0.3);
    ctx.fillStyle = '#05070d';
    sky.skyline.forEach((b) => {
      const bx = b.x * w, bw = b.w * w;
      const bh = b.h * h * (0.45 + 0.55 * k);
      const top = hy - bh;
      if (b.broken){                                         // выломанная верхушка
        ctx.fillRect(bx, top, bw * 0.55, bh);
        ctx.fillRect(bx + bw * 0.55, top + bh * 0.38, bw * 0.45, bh * 0.62);
      } else {
        ctx.fillRect(bx, top, bw, bh);
      }
    });
    ctx.restore();
  }

  function skyRender(){
    const ctx = sky.ctx, w = sky.w, h = sky.h;
    const cam = skyCam();
    ctx.clearRect(0, 0, w, h);

    const bg = ctx.createLinearGradient(0, 0, 0, h);
    bg.addColorStop(0, '#04050b');
    bg.addColorStop(1, '#080a15');
    ctx.fillStyle = bg;
    ctx.fillRect(0, 0, w, h);

    // Туманности в тех же двух цветах, что и название игры — не случайный третий акцент.
    const neb = (x, y, r, col) => {
      const g = ctx.createRadialGradient(x, y, 0, x, y, r);
      g.addColorStop(0, col);
      g.addColorStop(1, 'rgba(0,0,0,0)');
      ctx.fillStyle = g;
      ctx.fillRect(0, 0, w, h);
    };
    const nebA = Math.max(0, 1 - cam.surf);                 // у земли туманностей не видно
    neb(w * 0.22, h * 0.18, Math.max(w, h) * 0.55, 'rgba(47,110,100,' + (0.20 * nebA) + ')');
    neb(w * 0.85, h * 0.82, Math.max(w, h) * 0.5, 'rgba(120,40,70,' + (0.16 * nebA) + ')');

    skyDrawStars(ctx, cam);

    // Солнце и дыра ЖИВУТ В СЦЕНЕ, а не приклеены к экрану: при наезде мы пролетаем
    // мимо них, поэтому они разъезжаются от центра и гаснут. Раньше они стояли на
    // фиксированных долях экрана и при зуме оставались на месте — это и выглядело
    // сломанным. push растёт со ступенью наезда.
    const push = 1 + sky.z * 1.9;
    const away = (fx, fy) => ({ x: w * 0.5 + (fx - 0.5) * w * push, y: h * 0.5 + (fy - 0.5) * h * push });
    const farFade = Math.max(0, 1 - sky.z * 0.55) * Math.max(0, 1 - cam.surf);
    const sunP = away(0.14, 0.16);
    const holeP = away(0.87, 0.17);
    const sunX = sunP.x, sunY = sunP.y;
    skyDrawBlackHole(ctx, holeP.x, holeP.y, Math.max(6, Math.min(w, h) * 0.019 * (1 + sky.z * 0.35)), farFade);
    skyDrawSun(ctx, sunX, sunY, Math.max(4, Math.min(w, h) * 0.012 * (1 + sky.z * 0.3)), farFade);

    // Свет для всех шаров — из реального положения солнца на экране.
    const ldx = sunX - cam.cx, ldy = sunY - cam.cy;
    const ll = Math.hypot(ldx, ldy) || 1;
    const lit = { x: ldx / ll, y: ldy / ll };

    // Луна на орбите: за планетой или перед ней — по фазе, это и даёт глубину.
    const mPhase = sky.t * 0.16;
    const mOrb = cam.R * 1.9;
    const mx = cam.cx + Math.cos(mPhase) * mOrb;
    const my = cam.cy + Math.sin(mPhase) * mOrb * 0.32;
    const mR = Math.max(2, cam.R * 0.14);
    const moonBehind = Math.sin(mPhase) < 0;
    const moonFade = Math.max(0, 1 - cam.surf * 1.4);
    if (moonBehind) skyDrawMoon(ctx, mx, my, mR, lit, moonFade);
    skyDrawPlanet(ctx, cam.cx, cam.cy, cam.R, lit, cam);
    if (!moonBehind) skyDrawMoon(ctx, mx, my, mR, lit, moonFade);

    skyDrawEyes(ctx, cam.cx, cam.cy, cam.R, Math.max(0, 1 - cam.surf * 1.6));
    skyDrawPins(ctx, cam.cx, cam.cy, cam.R, cam.pin);
    skyDrawSurface(ctx, cam.surf, cam.cy - cam.R);          // горизонт = верх сферы

    // Буфер на экран: растяжение без сглаживания и даёт пиксельную картинку.
    sky.out.imageSmoothingEnabled = false;
    sky.out.clearRect(0, 0, sky.cssW, sky.cssH);
    sky.out.drawImage(sky.buf, 0, 0, sky.cssW, sky.cssH);
  }

  function skyFrame(now){
    sky.raf = requestAnimationFrame(skyFrame);
    const dt = Math.min(0.05, (now - sky.last) / 1000) || 0;
    sky.last = now;
    sky.t += dt;
    // Событие resize приходит не всегда (на телефоне адресная строка меняет высоту
    // без него) — сверяем фактический размер каждый кадр, это дешевле одного бага.
    if (sky.canvas.clientWidth !== sky.cssW || sky.canvas.clientHeight !== sky.cssH) skyResize();
    sky.z += (sky.zTarget - sky.z) * (SKY_REDUCED ? 1 : 1 - Math.pow(0.04, dt));
    if (!SKY_REDUCED){
      if (sky.spinTarget !== null && sky.z > 0.35){
        // Доворот к выбранной локации кратчайшей дугой.
        const d = ((sky.spinTarget - sky.spin + 540) % 360) - 180;
        sky.spin += d * (1 - Math.pow(0.02, dt));
      } else {
        sky.spin = (sky.spin + dt * 2.4) % 360;
      }
    }
    skyRender();
  }

  function skyStart(){
    if (!sky.ready) skyInit();
    if (!sky.ready || sky.raf) return;
    sky.last = performance.now();
    sky.raf = requestAnimationFrame(skyFrame);
  }
  function skyStop(){
    if (sky.raf){ cancelAnimationFrame(sky.raf); sky.raf = 0; }
  }
  function skyZoom(stage){
    sky.zTarget = stage;
    document.getElementById('mainMenu').classList.toggle('zoomed', stage > 0);
    if (SKY_REDUCED) sky.z = stage;
  }
